#include <windows.h>
#include <iostream>

#define TEST_HOTKEY 1
#define DISPLAY_HOTKEY 2
#define ENUM_DISPLAYS_HOTKEY 3 

constexpr int test_hotkey_id = TEST_HOTKEY;
constexpr int display_hotkey_id = DISPLAY_HOTKEY;
constexpr int enum_displays_hotkey_id = ENUM_DISPLAYS_HOTKEY;

constexpr UINT VM_HOTKEY1_TRIGGERED = WM_APP + 1;

bool clean_up(HWND hwnd) {
	BOOL res1 = UnregisterHotKey(hwnd, TEST_HOTKEY);
	BOOL res2 = UnregisterHotKey(hwnd, DISPLAY_HOTKEY);
	BOOL res3 = UnregisterHotKey(hwnd, ENUM_DISPLAYS_HOTKEY);
	return res1 && res2 && res3;
}
BOOL __stdcall monitor_process(HMONITOR hmonitor, HDC hdc, LPRECT rect, LPARAM data) {
	std::cout
		<< "pos: " << rect->left
		<< ", " << rect->top << '\n'
		<< "res: "
		<< rect->right - rect->left
		<< " x "
		<< rect->bottom - rect->top
		<< '\n';

	return TRUE;
}
LRESULT __stdcall message_process(HWND hwnd, UINT m, WPARAM wp, LPARAM lp) {
	switch (m) {
	case WM_DESTROY:
		if (!clean_up(hwnd)) {
			std::cerr << "cleanup failed\n";
		}
		PostQuitMessage(0);
		return 0;
	case WM_HOTKEY:
		if (wp == TEST_HOTKEY) { 
			std::cout << "test_hotkey_id triggered\n";

			PostMessageW(hwnd, VM_HOTKEY1_TRIGGERED, NULL, NULL);
			
			return 0;
		}
		if (wp == DISPLAY_HOTKEY) {
			HMONITOR hmonitor = MonitorFromWindow(hwnd, MONITOR_DEFAULTTONEAREST);

			MONITORINFO monitor_info{};

			monitor_info.cbSize = sizeof(monitor_info);

			if (GetMonitorInfoW(hmonitor, &monitor_info)) {
				RECT rect = monitor_info.rcMonitor;
				LONG width = rect.right - rect.left;
				LONG height = rect.bottom - rect.top;

				std::cout << "width: " << width << "\nheight: " << height << "\n";
			}
			else {
				std::cerr << "getting monitor info failed";
			}

			return 0;
		}
		if (wp == ENUM_DISPLAYS_HOTKEY) {
			if (!(EnumDisplayMonitors(nullptr, nullptr, monitor_process, NULL)))
				std::cerr << "enum display monitor failed";
			return 0;
		}
		break;
	case VM_HOTKEY1_TRIGGERED:
		std::cout << "test_hotkey_id message triggered\n";

		return 0;
	}

	return DefWindowProcW(hwnd, m, wp, lp);
}

void init_window_class(WNDCLASS* window_class, HINSTANCE instance) {
	window_class->lpfnWndProc = message_process;
	window_class->hInstance = instance;
	window_class->lpszClassName = L"my_window";
}
int main() {

	WNDCLASSW window_class{};

	HINSTANCE instance = GetModuleHandleW(nullptr);

	init_window_class(&window_class, instance);

	ATOM result = RegisterClassW(&window_class);

	if (!result) {
		std::cerr << "window registration failed: " << GetLastError() << "\n";
		return 1;
	}

	constexpr LPCWSTR window_name = L"test window";

	HWND hwnd = CreateWindowExW(0, window_class.lpszClassName, window_name, WS_OVERLAPPEDWINDOW, CW_USEDEFAULT, CW_USEDEFAULT, 800, 600, nullptr, nullptr, instance, nullptr);

	if (!hwnd) {
		std::cerr << "window creation failed: " << GetLastError() << "\n";

		return 1;
	}

	RegisterHotKey(hwnd, test_hotkey_id, MOD_CONTROL | MOD_ALT | MOD_NOREPEAT , 'F');
	RegisterHotKey(hwnd, display_hotkey_id, MOD_CONTROL | MOD_ALT | MOD_NOREPEAT, 'V');
	RegisterHotKey(hwnd, enum_displays_hotkey_id, MOD_CONTROL | MOD_ALT | MOD_NOREPEAT, 'B');

	ShowWindow(hwnd, SW_SHOW);

	MSG m{};

	while (true) { 

		BOOL result = GetMessageW(&m, nullptr, NULL, NULL);

		if (result == -1)
			return 1;
		if (result == 0)
			break;

		TranslateMessage(&m);
		DispatchMessage(&m);
	}	

	return 0;
}
